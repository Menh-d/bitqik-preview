1. About Us: Marquee Along SVG Path Component

Embed this code into the About Us page below the text/image section for the smooth floating crypto coins animation along an SVG wave path.

Notes for IT: Set `baseVelocity: 3.0` for slow smooth floating effect. Works on all modern browsers. 
<!-- ========================================== -->
<!-- ABOUT US: MARQUEE ALONG SVG PATH COMPONENT  -->
<!-- ========================================== -->
<div class="bq-white-path-wrapper" id="bq-marquee-path-studio" style="background:#ffffff; padding: 60px 0; overflow:hidden; width:100%;">
  <style>
    .bq-white-path-container { position: relative; width: 100%; max-width: 1200px; margin: 0 auto; height: 350px; }
    .bq-white-path-svg { position: absolute; top: 0; left: 0; width: 100%; height: 100%; pointer-events: none; }
    .bq-clean-coin { width: 52px; height: 52px; border-radius: 50%; display: flex; align-items: center; justify-content: center; background: #ffffff; box-shadow: 0 8px 24px rgba(0,0,0,0.08); transition: transform 0.3s ease; cursor: pointer; }
    .bq-clean-coin:hover { transform: scale(1.3); z-index: 10; }
    .bq-clean-coin img { width: 42px; height: 42px; object-fit: contain; border-radius: 50%; }
  </style>

  <div class="bq-white-path-container" id="bq-marquee-container">
    <svg class="bq-white-path-svg" viewBox="0 0 1200 350" preserveAspectRatio="none">
      <path id="bq-svg-path-clean" d="M -50 210 C 150 290, 350 310, 520 180 C 680 50, 850 80, 1020 220 C 1120 290, 1220 250, 1300 200" fill="none" stroke="transparent" stroke-width="2"/>
      <g id="bq-coins-layer"></g>
    </svg>
  </div>
</div>

<script>
(function() {
  const coinIcons = [
    'https://cryptologos.cc/logos/bitcoin-btc-logo.png',
    'https://cryptologos.cc/logos/ethereum-eth-logo.png',
    'https://cryptologos.cc/logos/tether-usdt-logo.png',
    'https://cryptologos.cc/logos/solana-sol-logo.png',
    'https://cryptologos.cc/logos/bnb-bnb-logo.png',
    'https://cryptologos.cc/logos/xrp-xrp-logo.png',
    'https://cryptologos.cc/logos/cardano-ada-logo.png',
    'https://cryptologos.cc/logos/dogecoin-doge-logo.png',
    'https://cryptologos.cc/logos/avalanche-avax-logo.png',
    'https://cryptologos.cc/logos/polkadot-new-dot-logo.png',
    'https://cryptologos.cc/logos/chainlink-link-logo.png',
    'https://cryptologos.cc/logos/near-protocol-near-logo.png',
    'https://cryptologos.cc/logos/tron-trx-logo.png'
  ];

  function initMarquee() {
    const path = document.getElementById('bq-svg-path-clean');
    const layer = document.getElementById('bq-coins-layer');
    if (!path || !layer) return;

    let pathLen = path.getTotalLength();
    if (!pathLen) { requestAnimationFrame(initMarquee); return; }

    layer.innerHTML = '';
    const items = [];
    const count = coinIcons.length;
    const spacing = pathLen / count;

    for (let i = 0; i < count; i++) {
      const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
      const fo = document.createElementNS('http://www.w3.org/2000/svg', 'foreignObject');
      fo.setAttribute('width', '60');
      fo.setAttribute('height', '60');
      fo.setAttribute('x', '-30');
      fo.setAttribute('y', '-30');

      const div = document.createElement('div');
      div.className = 'bq-clean-coin';
      div.innerHTML = '<img src="' + coinIcons[i] + '" alt="crypto">';
      fo.appendChild(div);
      g.appendChild(fo);
      layer.appendChild(g);

      items.push({ el: g, dist: i * spacing });
    }

    let lastTime = performance.now();
    const speed = 40; // pixels per second

    function animate(now) {
      const dt = (now - lastTime) / 1000;
      lastTime = now;

      items.forEach(item => {
        item.dist = (item.dist + speed * dt) % pathLen;
        const pt = path.getPointAtLength(item.dist);
        item.el.setAttribute('transform', 'translate(' + pt.x + ',' + pt.y + ')');
      });

      requestAnimationFrame(animate);
    }
    requestAnimationFrame(animate);
  }

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initMarquee);
  } else {
    initMarquee();
  }
})();
</script>

<script>
document.addEventListener('DOMContentLoaded', function() {
  const stack = document.querySelector('.careers-image-stack');
  const cards = stack ? stack.querySelectorAll('div') : [];
  const lightbox = document.getElementById('careers-lightbox');
  const lightboxImg = document.getElementById('careers-lightbox-img');
  const closeBtn = document.querySelector('.careers-lightbox-close');

  cards.forEach((card, index) => {
    const angle = (index - 3) * 8;
    const translateX = (index - 3) * 45;
    const translateY = Math.abs(index - 3) * 10;
    card.style.transform = `translateX(${translateX}px) translateY(${translateY}px) rotate(${angle}deg)`;
    card.style.zIndex = index + 1;

    card.addEventListener('click', function() {
      const img = card.querySelector('img');
      if (img && lightbox && lightboxImg) {
        lightboxImg.src = img.src;
        lightbox.classList.add('active');
      }
    });
  });

  if (closeBtn && lightbox) {
    closeBtn.addEventListener('click', () => lightbox.classList.remove('active'));
    lightbox.addEventListener('click', (e) => { if (e.target === lightbox) lightbox.classList.remove('active'); });
  }
});
</script>
